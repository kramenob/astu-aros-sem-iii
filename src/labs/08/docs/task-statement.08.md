Дан одномерный массив из 10 целых чисел:

Найти сумму элементов массива.

Код программы на С++ с ассемблерной вставкой:

```cpp
#include \<iostream\>

#include \<conio.h\>

using namespace std;

int main()

{

// ЗАДАНИЕ № 0:

// Дан одномерный массив из 10 целых чисел:

// 5, 10, 2, 4, 5, 7, 8, 10, -5, 2

// Нужно найти сумму элементов массива.

// Ожидаемый результат 48.

setlocale(LC_ALL, \"Russian\"); // русский алфавит

int x\[\] = { 5, 10, 2, 4, 5, 7, 8, 10, -5, 2 }; // исходный массив,
каждый элемент по 4 байта!!!

int xsize = sizeof(x)/sizeof(int); // вычисление количества элементов
массива

int sum = 0, i = 0; //sum - сумма элементов, i - параметр цикла

// Примечание: в Visual Studio целое число типа int занимает 4 байта.

\_\_asm

{

lea esi, x // загрузка в индексный регистр ESI адрес массива x

mov eax, 0 // обнуление сумматора EAX

mov ecx, dword ptr xsize // загрузка в регистр EСX размера массива

metka:

add eax, \[esi\] // добавить в сумматор EAX следующий элемент массива

add esi, 4 // передвигаемся к следующему элементу (+ 4 байта)

dec ecx // вычитание единицы из регистра EСX

jnz metka // проверка на нуль. Если не нуль, то переход на метку n

mov dword ptr sum, eax // исчерпаны все элементы массива. Выгрузка в sum
содержимого регистра EAX

}

printf(\"Иванов Иван. Лабораторная работа №10. Циклы и массивы. Вариант
№ 0\\n\\n\");

printf(\"Assembler. Сумма элементов массива равна: %d\", sum);

printf(\"\\n\");

sum=0;

for(i=0; i\<xsize; i++)

{

sum = sum + x\[i\];

}

printf(\"C++. Сумма элементов массива равна: %d\", sum);

printf(\"\\n\\n\");

system(\"pause\");

return 0;

}

```
